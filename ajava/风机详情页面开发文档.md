## 🔌 后端API接口需求

### 1. **获取设备基本信息**
```javascript
GET /sw/turbine/getDeviceInfo
参数: { turbineName: string }
返回: {
  deviceName: string,      // 设备名称
  deviceNo: string,        // 设备编号
  manufacturer: string,    // 制造厂商
  brakeType: string,       // 制动器类型
  ratedLoad: string,       // 额定载重
  ratedSpeed: string,      // 额定速度
  ratedPower: string,      // 额定功率
  productionDate: string,  // 生产日期
  useUnit: string,         // 使用单位
  installDate: string,     // 安装日期
  manager: string          // 管理人员
}
```

### 2. **获取报警历史**
```javascript
GET /sw/turbine/getAlarmHistory
参数: { 
  turbineName: string,
  pageNum?: number,    // 可选，默认1
  pageSize?: number    // 可选，默认10
}
返回: [{
  time: string,        // 时间 "2021-09-11 14:06"
  type: string,        // 问题类型 "叶轮振动异常"
  level: string        // 级别 "高" | "中" | "低"
}]
```

### 3. **获取故障统计**
```javascript
GET /sw/turbine/getFaultStats
参数: { 
  turbineName: string,
  startTime?: string,  // 可选，开始时间
  endTime?: string     // 可选，结束时间
}
返回: {
  totalCount: number,  // 总故障次数
  faultTypes: [{
    name: string,      // 故障类型名称
    count: number,     // 发生次数
    color: string      // 前端用于显示的颜色（可选，前端也可自行定义）
  }]
}
```

### 4. **获取边缘端信息**
```javascript
GET /sw/turbine/getEdgeInfo
参数: { turbineName: string }
返回: {
  cpuUsage: number,      // CPU利用率 0-100
  memoryUsage: number,   // 内存占用 0-100
  gpuUsage: number,      // GPU利用率 0-100
  diskUsage: number,     // 硬盘空间 0-100
  rttValue: number,      // 往返时延(ms)
  packetLoss: number     // 丢包率 0-100
}
```

### 5. **获取特征趋势数据**
```javascript
GET /sw/turbine/getTrendData
参数: { 
  turbineName: string,
  featureType?: string,  // 特征类型，如 "vibration"、"temperature"
  timeRange?: string     // 时间范围，如 "1h"、"24h"、"7d"
}
返回: {
  xData: string[],       // X轴数据（时间点或序号）
  series: [{
    name: string,        // 系列名称
    data: number[]       // Y轴数据
  }]
}
```

### 6. **获取传感器测点数据**
```javascript
GET /sw/turbine/getSensorPoints
参数: { turbineName: string }
返回: [{
  pointId: string,       // 测点ID
  pointName: string,     // 测点名称 "发电机A"
  componentType: string, // 部件类型 "generator"、"gearbox"、"bearing"
  position: {            // 3D模型上的位置（百分比）
    top: string,         // "20%"
    left: string         // "15%"
  },
  currentValue: number,  // 当前值
  status: string         // 状态 "normal" | "warning" | "alarm"
}]
```

### 7. **获取报警/故障次数分析**
```javascript
GET /sw/turbine/getAlarmFaultAnalysis
参数: { 
  turbineName: string,
  type: string          // "alarm" | "fault"
}
返回: {
  xData: string[],      // X轴数据（时间段）
  normalData: number[], // 正常数据
  abnormalData: number[] // 异常数据
}
```

## 🔄 数据流程

```
用户点击风机图标
    ↓
携带参数跳转到详情页
    ↓
详情页 mounted 钩子触发
    ↓
并行请求7个API接口：
    ├─ getDeviceInfo()          → deviceInfo
    ├─ getAlarmHistory()        → alarmHistory
    ├─ getFaultStats()          → faultTypes, totalFaultCount
    ├─ getEdgeInfo()            → CPU/内存/GPU/硬盘等数据
    ├─ getTrendData()           → 趋势图数据
    ├─ getSensorPoints()        → 传感器测点位置
    └─ getAlarmFaultAnalysis()  → 报警分析柱状图数据
    ↓
渲染页面和图表
```
